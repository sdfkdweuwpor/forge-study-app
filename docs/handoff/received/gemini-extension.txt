export const PROTOCOL_VERSION = 1;

export interface ScheduleWindow {
  days: number[]; // 0=Sun
  start: string;  // 'HH:mm'
  end: string;    // 'HH:mm'
}

export interface BlockerConfig {
  mode: 'focus' | 'schedule' | 'always';
  blocklist: string[];
  allowlist: string[];
  schedule: ScheduleWindow[];
  motivation: string[];
}

export interface SessionState {
  active: boolean;
  endsAt: number | null;
  taskTitle: string | null;
}

export interface BlockEvent {
  id: string;
  at: number;
  kind: 'blocked' | 'unlock';
  domain: string;
  unlockMinutes?: number;
}

export type AppMessage =
  | { v: 1; type: 'ping' }
  | { v: 1; type: 'sync'; config: BlockerConfig }
  | { v: 1; type: 'session'; session: SessionState | null }
  | { v: 1; type: 'getEvents'; since: number };

export type AppResponse =
  | { ok: true; version?: string; events?: BlockEvent[]; cursor?: number }
  | { ok: false; error: string };

export function isAppMessage(x: unknown): x is AppMessage {
  if (typeof x !== 'object' || x === null) return false;
  const msg = x as Record<string, unknown>;
  if (msg['v'] !== PROTOCOL_VERSION) return false;
  return (
    msg['type'] === 'ping' ||
    msg['type'] === 'sync' ||
    msg['type'] === 'session' ||
    msg['type'] === 'getEvents'
  );
}

    ],

    schedule: [],
    motivation: ['Get back to work.', 'Stay focused on what matters.']
  },

  dailyAttempts: {
    date: new Date().toISOString().split('T')[0] ?? '',
    count: 0

export async function getState(): Promise<ExtensionState> {
  const data = await chrome.storage.local.get(null);
  return {
    ...DEFAULT_STATE,
    ...data,
    config: {
      ...DEFAULT_STATE.config,
      ...((data['config'] as Partial<BlockerConfig>) || {})
    },
    dailyAttempts: {
      ...DEFAULT_STATE.dailyAttempts,
      ...((data['dailyAttempts'] as { date: string; count: number }) || {})
    }
  } as ExtensionState;
}

export async function setState(updates: Partial<ExtensionState>): Promise<void> {
  await chrome.storage.local.set(updates);
}

export async function addEvent(event: BlockEvent): Promise<void> {
  const state = await getState();
  const events = [...state.events, event].slice(-5000);

  const today = new Date().toISOString().split('T')[0] ?? '';
  let dailyAttempts = { ...state.dailyAttempts };

  if (dailyAttempts.date !== today) {
    dailyAttempts = { date: today, count: 0 };
  }
  if (event.kind === 'blocked') {
    dailyAttempts.count += 1;
  }

  await setState({ events, dailyAttempts });
}

import { ExtensionState } from './state.js';

export function isScheduleActive(
  schedule: ExtensionState['config']['schedule'],
  nowMs: number
): boolean {
  if (schedule.length === 0) return false;
  const d = new Date(nowMs);
  const day = d.getDay();
  const hhmm = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;

  return schedule.some((window) => {
    if (!window.days.includes(day)) return false;
    if (window.start <= window.end) {
      return hhmm >= window.start && hhmm < window.end;
    }
    // Overnight window (e.g. 22:00 -> 06:00)
    return hhmm >= window.start || hhmm < window.end;
  });
}

export function buildRules(
  state: ExtensionState,
  nowMs: number,
  extensionUrl: string
): chrome.declarativeNetRequest.Rule[] {
  const { config, session, unlocks } = state;
  let shouldBlock = false;

  if (config.mode === 'always') {
    shouldBlock = true;
  } else if (config.mode === 'focus') {
    shouldBlock = Boolean(session?.active);
  } else if (config.mode === 'schedule') {
    shouldBlock = isScheduleActive(config.schedule, nowMs);
  }

  if (!shouldBlock) return [];

  let nextId = 1;
  const rules: chrome.declarativeNetRequest.Rule[] = [];

  // 1. Block Rules
  for (const rawDomain of config.blocklist) {
    const domain = rawDomain.trim().toLowerCase();
    if (!domain) continue;

    const activeUnlock = unlocks.find((u) => u.domain === domain && u.until > nowMs);
    if (activeUnlock) continue;

    const escapedDomain = domain.replace(/\./g, '\\.');
    rules.push({
      id: nextId++,
      priority: 1,
      action: {
        type: chrome.declarativeNetRequest.RuleActionType.REDIRECT,
        redirect: {
          regexSubstitution: `${extensionUrl}blocked.html?site=${encodeURIComponent(domain)}&url=\\0`
        }
      },
      condition: {
        regexFilter: `^https?://([^/]+\\.)?${escapedDomain}/.*`,
        resourceTypes: [chrome.declarativeNetRequest.ResourceType.MAIN_FRAME]
      }
    });
  }

  // 2. Allowlist Rules (Higher Priority)
  for (const rawPrefix of config.allowlist) {
    const prefix = rawPrefix.trim();
    if (!prefix) continue;

    rules.push({
      id: nextId++,
      priority: 2,
      action: {
        type: chrome.declarativeNetRequest.RuleActionType.ALLOW
      },
      condition: {
        urlFilter: `||${prefix}`,
        resourceTypes: [chrome.declarativeNetRequest.ResourceType.MAIN_FRAME]
      }
    });
  }

  return rules;
}

import { describe, it, expect } from 'vitest';
import { buildRules, isScheduleActive } from './rules.js';
import { ExtensionState } from './state.js';

describe('buildRules and Schedule Matching', () => {
  const baseConfig: ExtensionState['config'] = {
    mode: 'always',
    blocklist: ['x.com', 'reddit.com'],
    allowlist: ['reddit.com/r/productivity'],
    schedule: [],
    motivation: []
  };

  const baseState: ExtensionState = {
    config: baseConfig,
    session: null,
    unlocks: [],
    events: [],
    dailyAttempts: { date: '2026-09-29', count: 0 }
  };

  const extUrl = 'chrome-extension://gpinhblnpebjbiodblihfpjbffacipbd/';

  it('builds block and allow rules in always mode', () => {
    const rules = buildRules(baseState, Date.now(), extUrl);
    expect(rules).toHaveLength(3); // 2 block, 1 allow

    const allowRule = rules.find((r) => r.priority === 2);
    expect(allowRule).toBeDefined();
    expect(allowRule?.condition.urlFilter).toBe('||reddit.com/r/productivity');

    const xRule = rules.find(
      (r) => r.condition.regexFilter?.includes('x\\.com') && r.priority === 1
    );
    expect(xRule).toBeDefined();
    expect(xRule?.action.redirect?.regexSubstitution).toContain('site=x.com');
  });

  it('does not produce rules in focus mode when inactive', () => {
    const state: ExtensionState = {
      ...baseState,
      config: { ...baseConfig, mode: 'focus' },
      session: { active: false, endsAt: null, taskTitle: null }
    };
    expect(buildRules(state, Date.now(), extUrl)).toHaveLength(0);
  });

  it('produces rules in focus mode when session is active', () => {
    const state: ExtensionState = {
      ...baseState,
      config: { ...baseConfig, mode: 'focus' },
      session: { active: true, endsAt: Date.now() + 60000, taskTitle: 'Work' }
    };
    expect(buildRules(state, Date.now(), extUrl).length).toBeGreaterThan(0);
  });

  it('skips blocked domains that have an active unlock', () => {
    const now = Date.now();
    const state: ExtensionState = {
      ...baseState,
      unlocks: [{ domain: 'x.com', until: now + 5000 }]
    };
    const rules = buildRules(state, now, extUrl);
    const xRule = rules.find((r) => r.condition.regexFilter?.includes('x\\.com'));
    expect(xRule).toBeUndefined();
  });

  it('handles standard daytime schedule windows', () => {
    const wednesdayNoon = new Date('2026-09-30T12:00:00').getTime(); // Day 3
    const schedule = [{ days: [3], start: '09:00', end: '17:00' }];
    expect(isScheduleActive(schedule, wednesdayNoon)).toBe(true);

    const wednesdayEvening = new Date('2026-09-30T18:00:00').getTime();
    expect(isScheduleActive(schedule, wednesdayEvening)).toBe(false);
  });

  it('handles overnight schedule windows', () => {
    const schedule = [{ days: [3], start: '22:00', end: '06:00' }];
    const nightTime = new Date('2026-09-30T23:30:00').getTime();
    const earlyMorning = new Date('2026-09-30T04:30:00').getTime();
    const afternoon = new Date('2026-09-30T14:00:00').getTime();

    expect(isScheduleActive(schedule, nightTime)).toBe(true);
    expect(isScheduleActive(schedule, earlyMorning)).toBe(true);
    expect(isScheduleActive(schedule, afternoon)).toBe(false);
  });
});

import { isAppMessage } from '../shared/protocol.js';
import { getState, setState, ExtensionState } from './state.js';
import { buildRules } from './rules.js';

const ALLOWED_ORIGINS = [
  'https://forge-study-app.netlify.app',
  'http://localhost',
  'http://127.0.0.1'
];

function isAllowedOrigin(url: string | undefined): boolean {
  if (!url) return false;
  return ALLOWED_ORIGINS.some((allowed) => url.startsWith(allowed));
}

export async function refreshDynamicRules(): Promise<void> {
  const state = await getState();
  const extUrl = `chrome-extension://${chrome.runtime.id}/`;
  const now = Date.now();
  const rules = buildRules(state, now, extUrl);

  const currentRules = await chrome.declarativeNetRequest.getDynamicRules();
  const removeRuleIds = currentRules.map((r) => r.id);

  await chrome.declarativeNetRequest.updateDynamicRules({
    removeRuleIds,
    addRules: rules
  });

  await scheduleNextAlarm(state, now);
}

async function scheduleNextAlarm(state: ExtensionState, now: number): Promise<void> {
  await chrome.alarms.clearAll();
  let nextAlarmTime: number | null = null;

  for (const unlock of state.unlocks) {
    if (unlock.until > now) {
      if (nextAlarmTime === null || unlock.until < nextAlarmTime) {
        nextAlarmTime = unlock.until;
      }
    }
  }

  if (state.session?.active && state.session.endsAt && state.session.endsAt > now) {
    if (nextAlarmTime === null || state.session.endsAt < nextAlarmTime) {
      nextAlarmTime = state.session.endsAt;
    }
  }

  // Periodic alarm fallback for schedule transitions
  if (state.config.mode === 'schedule') {
    const oneMinuteLater = now + 60 * 1000;
    if (nextAlarmTime === null || oneMinuteLater < nextAlarmTime) {
      nextAlarmTime = oneMinuteLater;
    }
  }

  if (nextAlarmTime !== null) {
    chrome.alarms.create('forge-refresh', { when: nextAlarmTime });
  }
}

chrome.runtime.onMessageExternal.addListener((msg, sender, sendResponse) => {
  if (!isAllowedOrigin(sender.url)) {
    sendResponse({ ok: false, error: 'Unauthorized origin' });
    return;
  }

  if (!isAppMessage(msg)) {
    sendResponse({ ok: false, error: 'Malformed or invalid message protocol' });
    return;
  }

  (async () => {
    try {
      if (msg.type === 'ping') {
        const manifest = chrome.runtime.getManifest();
        sendResponse({ ok: true, version: manifest.version });
        return;
      }

      if (msg.type === 'sync') {
        await setState({ config: msg.config });
        await refreshDynamicRules();
        sendResponse({ ok: true });
        return;
      }

      if (msg.type === 'session') {
        await setState({ session: msg.session });
        await refreshDynamicRules();
        sendResponse({ ok: true });
        return;
      }

      if (msg.type === 'getEvents') {
        const state = await getState();
        const filtered = state.events.filter((e) => e.at > msg.since);
        sendResponse({ ok: true, events: filtered, cursor: Date.now() });
        return;
      }

      sendResponse({ ok: false, error: 'Unsupported message type' });
    } catch (err) {
      sendResponse({ ok: false, error: String(err) });
    }
  })();

  return true;
});

chrome.alarms.onAlarm.addListener(() => {
  void refreshDynamicRules();
});

chrome.storage.onChanged.addListener((_changes, area) => {
  if (area === 'local') {
    void refreshDynamicRules();
  }
});

chrome.runtime.onInstalled.addListener(() => {
  void refreshDynamicRules();
});

const views = {
  blocked: document.getElementById('view-blocked'),

      activateView('override');

overrideInput?.addEventListener('input', async () => {
  if (overrideInput.value.trim() === REQUIRED_PHRASE) {
    overrideInput.disabled = true;
    const now = Date.now();
    const unlockDurationMs = 5 * 60 * 1000;
    const state = await getState();

    const filteredUnlocks = state.unlocks.filter((u) => u.domain !== blockedSite);
    filteredUnlocks.push({ domain: blockedSite, until: now + unlockDurationMs });

    await setState({ unlocks: filteredUnlocks });
    await addEvent({
      id: crypto.randomUUID(),
      at: now,
      kind: 'unlock',
      domain: blockedSite,
      unlockMinutes: 5
    });

    window.location.href = originalUrl || `https://${blockedSite}`;
  }
});

async function initPage(): Promise<void> {
  const state = await getState();

  await addEvent({
    id: crypto.randomUUID(),
    at: Date.now(),
    kind: 'blocked',
    domain: blockedSite
  });

  const taskBadge = document.getElementById('task-badge');
  if (taskBadge && state.session?.taskTitle) {
    taskBadge.innerText = `Focus Task: ${state.session.taskTitle}`;
  }

  const quoteEl = document.getElementById('motivation-text');
  if (quoteEl && state.config.motivation.length > 0) {
    const list = state.config.motivation;
    const randomIndex = Math.floor(Math.random() * list.length);
    const chosenQuote = list[randomIndex];
    if (chosenQuote) {
      quoteEl.innerText = `"${chosenQuote}"`;
    }
  }

  const timerEl = document.getElementById('session-timer');
  if (timerEl && state.session?.active && state.session.endsAt) {
    const renderClock = () => {
      const remainingMs = Math.max(0, state.session!.endsAt! - Date.now());
      const mins = Math.floor(remainingMs / 60000);
      const secs = Math.floor((remainingMs % 60000) / 1000);
      timerEl.innerText = `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
    };
    renderClock();
    setInterval(renderClock, 1000);
  }
}

import crypto from 'crypto';

export function deriveExtensionId(manifestKey) {
  const binaryKey = Buffer.from(manifestKey, 'base64');
  const hash = crypto.createHash('sha256').update(binaryKey).digest('hex');
  const first32 = hash.slice(0, 32);

  return first32
    .split('')
    .map((char) => {
      const code = char.charCodeAt(0);
      // hex 0-9 -> charCode 48-57 maps to a-j (code + 49)
      // hex a-f -> charCode 97-102 maps to k-p (code + 10)
      return String.fromCharCode(code >= 97 ? code + 10 : code + 49);
    })
    .join('');
}

# Forge Focus Chrome Extension

A calm, Manifest V3 site blocker built for the Forge study application.

## Development & Build

1. **Build the extension**:
   ```bash
   npm run build:ext
   ```
   This compiles TypeScript via `tsc -p extension/tsconfig.json`, mirrors styling tokens and the Inter font, and rasterizes `extension/icons/icon.svg` into PNG assets at 16, 32, 48, and 128 px using Playwright. Output is placed in `extension/dist/`.

2. **Package zip for release**:
   ```bash
   npm run zip:ext
   ```
   Produces `forge-extension.zip` at the repository root.

3. **Run unit tests**:
   ```bash
   npm test -- --run
   ```

4. **Load unpacked into Chrome**:
   - Go to `chrome://extensions`
   - Enable "Developer mode" (top right)
   - Click "Load unpacked" and select `extension/dist`
   - Confirm the extension ID shows: `gpinhblnpebjbiodblihfpjbffacipbd`

## Message Protocol (App <-> Extension)

The extension accepts external connections only from:
- `https://forge-study-app.netlify.app/*`
- `http://localhost/*`
- `http://127.0.0.1/*`

All messages carry `v: 1` as defined in `extension/src/shared/protocol.ts`.
```typescript
chrome.runtime.sendMessage(EXTENSION_ID, { v: 1, type: 'ping' });
```
